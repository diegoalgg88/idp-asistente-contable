```python
    def obtener_uma_vigente(self, fecha: date = None) -> ValorUMA:
        """
        Obtiene el valor de UMA vigente para una fecha.
        
        Args:
            fecha: Fecha de vigencia (default: hoy)
            
        Returns:
            ValorUMA vigente para la fecha
        """
        if fecha is None:
            fecha = date.today()
        
        for uma in self.valores_uma:
            if uma.vigencia_desde <= fecha <= uma.vigencia_hasta:
                return uma
        
        # Si no encuentra, retornar el último valor
        return self.valores_uma[-1]
    
    def obtener_salario_minimo_vigente(self, fecha: date = None, zona: str = 'general') -> ValorSalarioMinimo:
        """
        Obtiene el valor de salario mínimo vigente para una fecha.
        
        Args:
            fecha: Fecha de vigencia (default: hoy)
            zona: Zona ('general' o 'fronteriza')
            
        Returns:
            ValorSalarioMinimo vigente para la fecha y zona
        """
        if fecha is None:
            fecha = date.today()
        
        for salario in self.valores_salarios:
            if (salario.vigencia_desde <= fecha <= salario.vigencia_hasta and 
                salario.zona == zona):
                return salario
        
        # Si no encuentra, retornar el último valor para la zona
        for salario in reversed(self.valores_salarios):
            if salario.zona == zona:
                return salario
        
        raise ValueError(f"No se encontró salario mínimo para zona {zona}")
    
    def calcular_factor_actualizacion(self, fecha_inicial: date, fecha_final: date) -> Decimal:
        """
        Calcula factor de actualización entre dos fechas.
        
        Args:
            fecha_inicial: Fecha inicial
            fecha_final: Fecha final
            
        Returns:
            Factor de actualización (ej. 1.1321 para 13.21% de inflación)
        """
        uma_inicial = self.obtener_uma_vigente(fecha_inicial)
        uma_final = self.obtener_uma_vigente(fecha_final)
        
        factor = uma_final.valor_diario / uma_inicial.valor_diario
        
        return factor.quantize(Decimal('0.0001'), rounding=ROUND_HALF_UP)
    
    def generar_proyecciones_2027(self, inpc_estimado: Decimal = Decimal('4.0')) -> Dict:
        """
        Genera proyecciones de valores para 2027.
        
        Args:
            inpc_estimado: INPC estimado para diciembre 2026 (default: 4.0%)
            
        Returns:
            Dict con proyecciones de valores

        salario_fronterizo_2026 = self.obtener_salario_minimo_vigente(date(2026, 12, 31), 'fronteriza')
        
        # Proyección UMA 2027
        uma_2027_diario = uma_2026.valor_diario * (1 + inpc_estimado / 100)
        uma_2027_mensual = uma_2027_diario * 30
        uma_2027_anual = uma_2027_diario * 365
        
        # Proyección salarios 2027 (estimación 12-14% para general, 5-6% para fronterizo)
        salario_general_2027 = salario_general_2026.valor_diario * Decimal('1.13')  # 13% estimado
        salario_fronterizo_2027 = salario_fronterizo_2026.valor_diario * Decimal('1.055')  # 5.5% estimado

            'uma_2027': {
```

### 4.1 Limitación 1: Publicación en Fechas Diferentes

**Problema:**
- UMA se publica en enero (vigencia 1-febrero)

### 4.2 Limitación 2: Proyecciones con Incertidumbre

**Problema:**
Las proyecciones de 2027 tienen incertidumbre (INPC estimado 3.5%-4.5%). Decisiones basadas en proyecciones pueden ser incorrectas.

**Solución:**
```python
def generar_escenarios_proyeccion(self, inpc_base: Decimal = Decimal('4.0')) -> Dict:
    """
    Genera tres escenarios de proyección (optimista, base, pesimista).
    
    Args:
        inpc_base: INPC estimado base (default: 4.0%)
        
    Returns:
        Dict con tres escenarios de proyección
    """
    escenarios = {
        'optimista': self.generar_proyecciones_2027(inpc_base - Decimal('0.5')),
        'base': self.generar_proyecciones_2027(inpc_base),
        'pesimista': self.generar_proyecciones_2027(inpc_base + Decimal('0.5'))
    }
    
    return escenarios

# Uso
```
