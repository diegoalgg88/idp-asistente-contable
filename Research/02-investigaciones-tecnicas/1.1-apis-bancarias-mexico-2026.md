## Implementación Técnica

### Funciones Python para Integración Bancaria

```python
"""
Módulo: integracion_bancaria_mexico.py
Propósito: Integración con APIs bancarias mexicanas (BBVA, Santander, STP, Plaid)
Versión: 1.0
Fecha: 2026-03-10
"""

import requests
from typing import Dict, List, Optional, Tuple
from datetime import datetime, timedelta
from dataclasses import dataclass
import base64
import hashlib


@dataclass
class CredencialesBanco:
    """Credenciales para conexión bancaria."""
    banco: str  # 'bbva', 'santander', 'citibanamex', 'banorte', 'stp'
    client_id: str
    client_secret: str
    api_key: Optional[str] = None
    certificado: Optional[str] = None
    llave_privada: Optional[str] = None


@dataclass
class TransaccionBancaria:
    """Representa una transacción bancaria."""
    id: str
    fecha: datetime
    descripcion: str
    monto: float
    moneda: str
    tipo: str  # 'CREDIT' o 'DEBIT'
    saldo: Optional[float] = None
    referencia: Optional[str] = None
    categoria: Optional[str] = None


class ClienteBancario:
    """
    Cliente base para conexión con APIs bancarias.
    """
    
    def __init__(self, credenciales: CredencialesBanco):
        """
        Inicializa el cliente bancario.
        
        Args:
            credenciales: Credenciales de autenticación
        """
        self.credenciales = credenciales
        self.access_token: Optional[str] = None
        self.token_expires_at: Optional[datetime] = None
        self.session = requests.Session()
    
    def autenticar(self) -> bool:
        """
        Realiza autenticación OAuth2.
        
        Returns:
            True si la autenticación fue exitosa
        """
        raise NotImplementedError("Subclases deben implementar autenticar()")
    
    def obtener_cuentas(self) -> List[Dict]:
        """
        Obtiene lista de cuentas del cliente.
        
        Returns:
            Lista de diccionarios con información de cuentas
        """
        raise NotImplementedError("Subclases deben implementar obtener_cuentas()")
    
    def obtener_transacciones(self, cuenta_id: str,
                               fecha_inicio: datetime,
                               fecha_fin: datetime) -> List[TransaccionBancaria]:
        """
        Obtiene transacciones de una cuenta.
        
        Args:
            cuenta_id: ID de la cuenta
            fecha_inicio: Fecha inicial
            fecha_fin: Fecha final
        
        Returns:
            Lista de TransaccionBancaria
        """
        raise NotImplementedError("Subclases deben implementar obtener_transacciones()")
    
    def _token_valido(self) -> bool:
        """
        Verifica si el token de acceso es válido.
        
        Returns:
            True si el token es válido y no ha expirado
        """
        if not self.access_token:
            return False
        
        if not self.token_expires_at:
            return True
        
        return datetime.now() < self.token_expires_at


class BBVACliente(ClienteBancario):
    """
    Cliente para API de BBVA Spark.
    """
    
    BASE_URL = "https://api.bbva.com"
    
    def autenticar(self) -> bool:
        """Autentica con BBVA Spark OAuth2."""
        url = f"{self.BASE_URL}/oauth2/token"
        
        credentials = f"{self.credenciales.client_id}:{self.credenciales.client_secret}"
        encoded = base64.b64encode(credentials.encode()).decode()
        
        headers = {
            'Authorization': f'Basic {encoded}',
            'Content-Type': 'application/x-www-form-urlencoded'
        }
        
        data = {
            'grant_type': 'client_credentials',
            'scope': 'accounts transactions'
        }
        
        try:
            response = self.session.post(url, headers=headers, data=data, timeout=30)
            response.raise_for_status()
            
            token_data = response.json()
            self.access_token = token_data['access_token']
            expires_in = token_data.get('expires_in', 3600)
            self.token_expires_at = datetime.now() + timedelta(seconds=expires_in)
            
            return True
            
        except requests.RequestException as e:
            print(f"Error de autenticación BBVA: {e}")
            return False
    
    def obtener_cuentas(self) -> List[Dict]:
        """Obtiene cuentas de BBVA."""
        if not self._token_valido():
            if not self.autenticar():
                return []
        
        url = f"{self.BASE_URL}/open-banking/v3.1/aisp/accounts"
        
        headers = {
            'Authorization': f'Bearer {self.access_token}',
            'x-fapi-financial-id': 'BBVAMX'
        }
        
        try:
            response = self.session.get(url, headers=headers, timeout=30)
            response.raise_for_status()
            
            data = response.json()
            return data.get('Data', {}).get('Account', [])
            
        except requests.RequestException as e:
            print(f"Error al obtener cuentas BBVA: {e}")
            return []
    
    def obtener_transacciones(self, cuenta_id: str,
                               fecha_inicio: datetime,
                               fecha_fin: datetime) -> List[TransaccionBancaria]:
        """Obtiene transacciones de cuenta BBVA."""
        if not self._token_valido():
            if not self.autenticar():
                return []
        
        url = f"{self.BASE_URL}/open-banking/v3.1/aisp/accounts/{cuenta_id}/transactions"
        
        headers = {
            'Authorization': f'Bearer {self.access_token}',
            'x-fapi-financial-id': 'BBVAMX'
        }
        
        params = {
            'fromBookingDateTime': fecha_inicio.isoformat(),
            'toBookingDateTime': fecha_fin.isoformat(),
            'limit': 100
        }
        
        try:
            response = self.session.get(url, headers=headers, params=params, timeout=30)
            response.raise_for_status()
            
            data = response.json()
            transacciones_raw = data.get('Data', {}).get('Transaction', [])
            
            transacciones = []
            for txn in transacciones_raw:
                monto_str = txn.get('Amount', {}).get('Amount', '0')
                monto = float(monto_str)
                
                tipo = 'CREDIT' if txn.get('CreditDebitIndicator') == 'Credit' else 'DEBIT'
                
                transacciones.append(TransaccionBancaria(
                    id=txn.get('TransactionId', ''),
                    fecha=datetime.fromisoformat(txn.get('BookingDateTime', '').replace('Z', '+00:00')),
                    descripcion=txn.get('MerchantName', txn.get('RemittanceInformation', '')),
                    monto=abs(monto),
                    moneda=txn.get('Amount', {}).get('Currency', 'MXN'),
                    tipo=tipo,
                    referencia=txn.get('TransactionId', '')
                ))
            
            return transacciones
            
        except requests.RequestException as e:
            print(f"Error al obtener transacciones BBVA: {e}")
            return []


class STPCliente(ClienteBancario):
    """
    Cliente para API de STP (SPEI).

    BASE_URL = "https://api.stp.com.mx"
    
    def autenticar(self) -> bool:
        """STP usa API Key en headers, no requiere autenticación OAuth2."""
        self.access_token = self.credenciales.api_key
        return True
    
    def crear_orden_spei(self, orden: Dict) -> Dict:
        """
        Crea orden de transferencia SPEI.
        
        Args:
            orden: Diccionario con datos de la orden
        
        Returns:
            Diccionario con resultado de la orden
        """
        url = f"{self.BASE_URL}/v1/ordenes"
        
        headers = {
            'Authorization': self.credenciales.api_key,
            'Content-Type': 'application/json'
        }
        
        try:
            response = self.session.post(url, headers=headers, json=orden, timeout=30)
            response.raise_for_status()
            
            return response.json()
            
        except requests.RequestException as e:
            print(f"Error al crear orden SPEI: {e}")
            return {'error': str(e)}
    
    def consultar_orden(self, orden_id: int) -> Dict:
        """
        Consulta estado de orden SPEI.

            return response.json()

class PlaidCliente(ClienteBancario):
    """
    Cliente para API de Plaid.
    """
    
    SANDBOX_URL = "https://sandbox.plaid.com"
    PRODUCTION_URL = "https://production.plaid.com"
    
    def __init__(self, credenciales: CredencialesBanco, ambiente: str = 'sandbox'):
        """
        Inicializa cliente Plaid.
        
        Args:
            credenciales: Credenciales (client_id, secret)
            ambiente: 'sandbox' o 'production'
        """
        super().__init__(credenciales)
        self.base_url = self.SANDBOX_URL if ambiente == 'sandbox' else self.PRODUCTION_URL
    
    def autenticar(self) -> bool:
        """Plaid usa API keys directamente en cada request."""
        self.access_token = self.credenciales.client_id
        return True
    
    def crear_link_token(self, cliente_nombre: str) -> str:
        """
        Crea link token para Plaid Link.
        
        Args:
            cliente_nombre: Nombre del cliente
        
        Returns:
            Link token para frontend
        """
        url = f"{self.base_url}/link/token/create"
        
        headers = {
            'Content-Type': 'application/json'
        }
        
        data = {
            'client_id': self.credenciales.client_id,
            'secret': self.credenciales.client_secret,
            'client_name': cliente_nombre,
            'country_codes': ['MX'],
            'language': 'es',
            'products': ['auth', 'transactions']
        }
        
        try:
            response = self.session.post(url, headers=headers, json=data, timeout=30)
            response.raise_for_status()
            
            return response.json().get('link_token', '')
            
        except requests.RequestException as e:
            print(f"Error al crear link token Plaid: {e}")
            return ''
    
    def intercambiar_public_token(self, public_token: str) -> str:
        """
        Intercambia public_token por access_token.
        
        Args:
            public_token: Token obtenido de Plaid Link
        
        Returns:
            Access token para API calls
        """
        url = f"{self.base_url}/item/public_token/exchange"
        
        headers = {
            'Content-Type': 'application/json'
        }
        
        data = {
            'client_id': self.credenciales.client_id,
            'secret': self.credenciales.client_secret,
            'public_token': public_token
        }
        
        try:
            response = self.session.post(url, headers=headers, json=data, timeout=30)
            response.raise_for_status()
            
            self.access_token = response.json().get('access_token', '')
            return self.access_token
            
        except requests.RequestException as e:
            print(f"Error al intercambiar token Plaid: {e}")
            return ''
    
    def obtener_transacciones(self, cuenta_id: str,
                               fecha_inicio: datetime,
                               fecha_fin: datetime) -> List[TransaccionBancaria]:
        """Obtiene transacciones de Plaid."""
        url = f"{self.base_url}/transactions/get"
        
        headers = {
            'Content-Type': 'application/json'
        }
        
        data = {
            'client_id': self.credenciales.client_id,
            'secret': self.credenciales.client_secret,
            'access_token': self.access_token,
            'start_date': fecha_inicio.strftime('%Y-%m-%d'),
            'end_date': fecha_fin.strftime('%Y-%m-%d')
        }
        
        try:
            response = self.session.post(url, headers=headers, json=data, timeout=30)
            response.raise_for_status()
            
            data = response.json()
            transacciones_raw = data.get('transactions', [])
            
            transacciones = []
            for txn in transacciones_raw:
                transacciones.append(TransaccionBancaria(
                    id=txn.get('transaction_id', ''),
                    fecha=datetime.strptime(txn.get('date', ''), '%Y-%m-%d'),
                    descripcion=txn.get('name', ''),
                    monto=abs(txn.get('amount', 0)),
                    moneda=txn.get('currency', 'MXN'),
                    tipo='CREDIT' if txn.get('amount', 0) < 0 else 'DEBIT',
                    categoria=txn.get('category', ['Uncategorized'])[0] if txn.get('category') else None
                ))
            
            return transacciones
            
        except requests.RequestException as e:
            print(f"Error al obtener transacciones Plaid: {e}")
            return []
    
    def obtener_cuentas(self) -> List[Dict]:
        """Obtiene cuentas de Plaid."""
        url = f"{self.base_url}/accounts/get"
        
        headers = {
            'Content-Type': 'application/json'
        }
        
        data = {
            'client_id': self.credenciales.client_id,
            'secret': self.credenciales.client_secret,
            'access_token': self.access_token
        }
        
        try:
            response = self.session.post(url, headers=headers, json=data, timeout=30)
            response.raise_for_status()
            
            return response.json().get('accounts', [])
            
        except requests.RequestException as e:
            print(f"Error al obtener cuentas Plaid: {e}")
            return []
```
