#### Schema de Datos

```python
from dataclasses import dataclass, field
from typing import List, Optional, Dict
from enum import Enum
import pandas as pd

class CategoriaNIF(Enum):
    OPERACION = "operacion"
    INVERSION = "inversion"
    FINANCIAMIENTO = "financiamiento"
    IMPUESTOS = "impuestos"
    ACTIVO = "activo"
    PASIVO = "pasivo"
    CAPITAL = "capital"

@dataclass
class EjemploClasificacion:
    """
    Ejemplo individual para entrenamiento de modelo de clasificación contable.
    
    Atributos:
        id: Identificador único del ejemplo
        concepto: Descripción del concepto del CFDI (ej: "Honorarios profesionales")
        proveedor: Nombre o RFC del proveedor
        monto: Monto de la transacción en MXN
        fecha: Fecha de la transacción (YYYY-MM-DD)
        tipo_cfdi: Tipo de comprobante (I=Ingreso, E=Egreso, T=Traslado, N=Nota crédito)
        metodo_pago: Método de pago (PUE=Pago único, PPD=Pago parcial)
        forma_pago: Forma de pago (01=Efectivo, 03=Transferencia, 04=Tarjeta)
        uso_cfdi: Uso del comprobante (G01=Gastos generales, G03=Gastos financieros)
        
        # Features derivadas
        concepto_normalizado: Concepto en minúsculas, sin acentos, sin puntuación
        monto_categoria: Categoría del monto (bajo <1000, medio 1000-10000, alto >10000)
        dia_semana: Día de la semana (0-6)
        mes: Mes del año (1-12)
        
        # Labels (etiquetas)
        cuenta_contable: Código de cuenta contable (ej: "604.01")
        codigo_sat: Código agrupador SAT (ej: "604.01")
        categoria_nif: Categoría NIF (operacion, inversion, financiamiento, etc.)
        deducible: Booleano indicando si es deducible de ISR
        iva_acreditable: Booleano indicando si el IVA es acreditable
        
        # Metadata
        confianza: Nivel de confianza en la etiqueta (1.0=seguro, 0.5=dudoso)
        fuente: Origen del dato (manual, automatico, sintético)
        validado: Booleano indicando si fue validado por contador certificado
    """
    id: str
    concepto: str
    proveedor: str
    monto: float
    fecha: str
    tipo_cfdi: str
    metodo_pago: str
    forma_pago: str
    uso_cfdi: str
    
    # Features derivadas (se calculan automáticamente)
    concepto_normalizado: str = field(init=False)
    monto_categoria: str = field(init=False)
    dia_semana: int = field(init=False)
    mes: int = field(init=False)
    
    # Labels
    cuenta_contable: str
    codigo_sat: str
    categoria_nif: CategoriaNIF
    deducible: bool = True
    iva_acreditable: bool = True
    
    # Metadata
    confianza: float = 1.0
    fuente: str = "manual"
    validado: bool = False
    
    def __post_init__(self):
        """Calcula features derivadas después de la inicialización."""
        self.concepto_normalizado = self._normalizar_texto(self.concepto)
        self.monto_categoria = self._categorizar_monto(self.monto)
        self.dia_semana = self._extraer_dia_semana(self.fecha)
        self.mes = self._extraer_mes(self.fecha)
    
    @staticmethod
    def _normalizar_texto(texto: str) -> str:
        """Normaliza texto: minúsculas, sin acentos, sin puntuación."""
        import unicodedata
        import re
        
        # Minúsculas
        texto = texto.lower()
        # Quitar acentos
        texto = ''.join(
            c for c in unicodedata.normalize('NFD', texto)
            if unicodedata.category(c) != 'Mn'
        )
        # Quitar puntuación
        texto = re.sub(r'[^\w\s]', '', texto)
        # Quitar espacios extra
        texto = ' '.join(texto.split())
        
        return texto
    
    @staticmethod
    def _categorizar_monto(monto: float) -> str:
        """Categoriza monto para feature engineering."""
        if monto < 1000:
            return "bajo"
        elif monto < 10000:
            return "medio"
        elif monto < 50000:
            return "alto"
        else:
            return "muy_alto"
    
    @staticmethod
    def _extraer_dia_semana(fecha: str) -> int:
        """Extrae día de la semana (0=lunes, 6=domingo)."""
        from datetime import datetime
        dt = datetime.strptime(fecha, '%Y-%m-%d')
        return dt.weekday()
    
    @staticmethod
    def _extraer_mes(fecha: str) -> int:
        """Extrae mes del año (1-12)."""
        from datetime import datetime
        dt = datetime.strptime(fecha, '%Y-%m-%d')
        return dt.month
    
    def to_dict(self) -> Dict:
        """Convierte a diccionario para DataFrame."""
        return {
            'id': self.id,
            'concepto': self.concepto,
            'concepto_normalizado': self.concepto_normalizado,
            'proveedor': self.proveedor,
            'monto': self.monto,
            'monto_categoria': self.monto_categoria,
            'fecha': self.fecha,
            'dia_semana': self.dia_semana,
            'mes': self.mes,
            'tipo_cfdi': self.tipo_cfdi,
            'metodo_pago': self.metodo_pago,
            'forma_pago': self.forma_pago,
            'uso_cfdi': self.uso_cfdi,
            'cuenta_contable': self.cuenta_contable,
            'codigo_sat': self.codigo_sat,
            'categoria_nif': self.categoria_nif.value,
            'deducible': self.deducible,
            'iva_acreditable': self.iva_acreditable,
            'confianza': self.confianza,
            'fuente': self.fuente,
            'validado': self.validado
        }


class DatasetContable:
    """
    Gestiona el dataset completo de entrenamiento para clasificación contable.
    
    Ejemplo:
        >>> dataset = DatasetContable()
        >>> dataset.cargar_desde_csv('dataset_entrenamiento.csv')
        >>> dataset.validar_calidad()
        >>> dataset.balancear_clases()
        >>> dataset.exportar_parquet('dataset_final.parquet')
    """
    
    def __init__(self):
        self.df: Optional[pd.DataFrame] = None
        self.estadisticas: Dict = {}
    
    def cargar_desde_csv(self, path: str) -> pd.DataFrame:
        """Carga dataset desde archivo CSV."""
        self.df = pd.read_csv(path)
        return self.df
    
    def validar_calidad(self) -> Dict:
        """
        Valida calidad del dataset.
        
        Returns:
            Diccionario con métricas de calidad
        """
        if self.df is None:
            raise ValueError("Dataset no cargado")
        
        metricas = {
            'total_ejemplos': len(self.df),
            'columnas': list(self.df.columns),
            'valores_nulos': self.df.isnull().sum().to_dict(),
            'clases_unicas': self.df['cuenta_contable'].nunique(),
            'distribucion_clases': self.df['cuenta_contable'].value_counts().to_dict(),
            'ejemplos_por_clase': self.df['cuenta_contable'].value_counts().min(),
            'clase_mas_frecuente': self.df['cuenta_contable'].mode()[0],
            'clase_menos_frecuente': self.df['cuenta_contable'].value_counts().idxmin(),
            'porcentaje_validados': (self.df['validado'] == True).mean() * 100 if 'validado' in self.df.columns else 0,
            'confianza_promedio': self.df['confianza'].mean() if 'confianza' in self.df.columns else 1.0
        }
        
        self.estadisticas = metricas
        return metricas
    
    def detectar_etiquetas_erroneas(self, umbral_confianza: float = 0.7) -> pd.DataFrame:
        """
        Detecta posibles etiquetas erróneas usando CleanLab.
        
        Args:
            umbral_confianza: Umbral para marcar ejemplo como sospechoso
            
        Returns:
            DataFrame con ejemplos sospechosos
        """
        try:
            from cleanlab.classification import CleanLearning
            from sklearn.feature_extraction.text import TfidfVectorizer
            from sklearn.ensemble import RandomForestClassifier
            
            # Preparar features
            vectorizer = TfidfVectorizer(max_features=1000)
            X = vectorizer.fit_transform(self.df['concepto_normalizado'])
            y = self.df['cuenta_contable']
            
            # Entrenar modelo base
            clf = RandomForestClassifier(n_estimators=100, random_state=42)
            clf.fit(X, y)
            
            # Usar CleanLab para detectar etiquetas erróneas
            cl = CleanLearning(clf=clf)
            label_issues = cl.find_label_issues(
                labels=y,
                pred_probs=clf.predict_proba(X),
                return_indices_ranked_by='self_confidence'
            )
            
            # Filtrar ejemplos sospechosos
            sospechosos = self.df.iloc[label_issues].copy()
            sospechosos['razon'] = 'Baja confianza del modelo'
            
            return sospechosos
            
        except ImportError:
            print("CleanLab no instalado. Instalar con: pip install cleanlab")
            return pd.DataFrame()
    
    def balancear_clases(self, metodo: str = 'smote') -> pd.DataFrame:
        """
        Balancea clases del dataset.
        
        Args:
            metodo: Método de balanceo ('smote', 'undersampling', 'oversampling', 'sintetico')
            
        Returns:
            DataFrame balanceado
        """
        from imblearn.over_sampling import SMOTE
        
        if metodo == 'smote':
            smote = SMOTE(random_state=42)
            X = self.df.drop(columns=['cuenta_contable', 'codigo_sat'])
            y = self.df['cuenta_contable']
            X_res, y_res = smote.fit_resample(X, y)
            self.df = pd.concat([X_res, y_res], axis=1)
        
        return self.df
    
    def exportar_parquet(self, path: str):
        """Exporta dataset a formato Parquet."""
        if self.df is None:
            raise ValueError("Dataset no cargado")
        self.df.to_parquet(path, index=False)
    
    def generar_reporte_calidad(self) -> str:
        """Genera reporte de calidad del dataset."""
        reporte = "## Reporte de Calidad del Dataset\n\n"
        reporte += f"**Total ejemplos:** {self.estadisticas['total_ejemplos']}\n"
        reporte += f"**Clases únicas:** {self.estadisticas['clases_unicas']}\n"
        reporte += f"**Ejemplos por clase (mín):** {self.estadisticas['ejemplos_por_clase']}\n"
        reporte += f"**Porcentaje validados:** {self.estadisticas['porcentaje_validados']:.1f}%\n"
        reporte += f"**Confianza promedio:** {self.estadisticas['confianza_promedio']:.2f}\n"
        
        return reporte
```

```python
def etiquetado_manual_lote(lote_cfdi: List[Dict], contador_id: str) -> List[EjemploClasificacion]:
```

```python
def auto_labeling_con_cleanlab(dataset: DatasetContable) -> DatasetContable:
    """
    Usa CleanLab para detectar y corregir etiquetas erróneas automáticamente.
    
    Proceso:
    1. Entrenar modelo inicial con dataset etiquetado manualmente
    2. CleanLab identifica ejemplos con etiquetas probablemente incorrectas
    3. Modelo sugiere correcciones
    4. Contador valida correcciones (solo casos dudosos)
    
    Resultado: 98% menos ejemplos que requieren revisión humana (caso Adm Cloud)
    """
    # Detectar etiquetas erróneas
    sospechosos = dataset.detectar_etiquetas_erroneas()
    
    print(f"Ejemplos sospechosos: {len(sospechosos)} de {len(dataset.df)}")
    print(f"Porcentaje a revisar: {len(sospechosos)/len(dataset.df)*100:.1f}%")
    
    # Para cada sospechoso, sugerir nueva etiqueta
    for idx, row in sospechosos.iterrows():
        # Modelo sugiere etiqueta alternativa
        etiqueta_sugerida = sugerir_etiqueta_alternativa(row)
        dataset.df.loc[idx, 'cuenta_contable_sugerida'] = etiqueta_sugerida
        dataset.df.loc[idx, 'confianza'] = 0.5  # Baja confianza para revisión
    
    return dataset
```

```python
def generar_ejemplos_sinteticos(conceptos_base: List[str], n_ejemplos: int = 1000) -> List[str]:
    """
    Genera conceptos sintéticos usando LLM para balancear dataset.
    
    Caso de uso: BMW generó 100,000 imágenes sintéticas con SORDI
    para reducir tiempo de etiquetado manual en 66%.
    
    Args:
        conceptos_base: Lista de conceptos reales (ej: "Honorarios profesionales")
        n_ejemplos: Número de ejemplos sintéticos a generar
        
    Returns:
        Lista de conceptos sintéticos
    """
    from openai import OpenAI
    
    client = OpenAI(api_key="tu-api-key")
    
    ejemplos_sinteticos = []
    
    for concepto in conceptos_base:
        prompt = f"""
        Genera 10 variaciones del siguiente concepto contable mexicano.
        Mantén el significado pero usa diferentes palabras, sinónimos o frases equivalentes.
        
        Concepto base: "{concepto}"
        
        Ejemplos:
        1. 
        2. 
        3. 
        ...
        10. 
        """
        
        response = client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": prompt}],
            max_tokens=500
        )
        
        variaciones = parse_variaciones(response.choices[0].message.content)
        ejemplos_sinteticos.extend(variaciones)
    
    return ejemplos_sinteticos[:n_ejemplos]
```

## 4. Limitantes y Restricciones

### 4.1 Limitación 1: Disponibilidad de Datos Reales

**Problema:**
Obtener 10,000+ CFDI reales etiquetados correctamente requiere:
- Acceso a datos de clientes reales (privacidad, NDAs)
- Tiempo de contadores certificados para etiquetado (20-30 horas)
- Validación de calidad (revisión de 10% del dataset)

**Solución:**
```python
def estrategia_dataset_hibrido() -> DatasetContable:
    """
    Combina datos reales, sintéticos y públicos.
    
    Composición recomendada:
    - 40% datos reales de clientes (4,000 ejemplos)
    - 30% datos sintéticos generados con LLM (3,000 ejemplos)
    - 20% datos de datasets públicos (2,000 ejemplos)
    - 10% datos aumentados con variaciones (1,000 ejemplos)
    """
    dataset = DatasetContable()
    
    # 1. Cargar datos reales (clientes que autorizaron)
    datos_reales = cargar_datos_reales(clientes_autorizados=['cliente1', 'cliente2'])
    
    # 2. Generar datos sintéticos basados en reales
    conceptos_base = datos_reales['concepto'].unique().tolist()
    datos_sinteticos = generar_ejemplos_sinteticos(conceptos_base, n_ejemplos=3000)
    
    # 3. Cargar datos públicos (si disponibles)
    datos_publicos = cargar_dataset_publico('mexican-accounting-dataset')
    
    # 4. Aumentar datos con variaciones
    datos_aumentados = aumentar_datos(datos_reales, variaciones=1000)
    
    # Combinar todo
    dataset.df = pd.concat([datos_reales, datos_sinteticos, datos_publicos, datos_aumentados])
    
    return dataset
```

**Impacto:**
- Reduce dependencia de datos reales en 60%
- Permite comenzar entrenamiento con dataset más pequeño
- Riesgo: datos sintéticos pueden no capturar patrones reales

### 4.2 Limitación 2: Etiquetas Inconsistentes entre Contadores

**Problema:**
Diferentes contadores pueden clasificar el mismo concepto de forma distinta:
- Contador A: "Honorarios profesionales" → 604.01 (Honorarios)
- Contador B: "Honorarios profesionales" → 604.38 (Honorarios personas morales)

Esto genera ruido en el dataset y reduce precisión del modelo.

**Solución:**
```python
def validar_consistencia_etiquetas(dataset: DatasetContable, n_validadores: int = 3) -> float:
    """
    Valida consistencia inter-validador (Inter-Annotator Agreement).
    
    Métrica: Kappa de Cohen (0-1, donde 1=acuerdo perfecto)
    
    Proceso:
    1. Seleccionar 100 ejemplos aleatorios
    2. 3 contadores etiquetan independientemente
    3. Calcular Kappa de Cohen
    4. Si Kappa < 0.7, revisar criterios de etiquetado
    
    Returns:
        Kappa de Cohen (0-1)
    """
    from sklearn.metrics import cohen_kappa_score
    
    # Seleccionar muestra
    muestra = dataset.df.sample(n=100, random_state=42)
    
    # Simular etiquetas de 3 validadores (en producción, esto sería etiquetado real)
    etiquetas_validador_1 = muestra['cuenta_contable']
    etiquetas_validador_2 = simular_etiquetas_con_ruido(etiquetas_validador_1, ruido=0.1)
    etiquetas_validador_3 = simular_etiquetas_con_ruido(etiquetas_validador_1, ruido=0.15)
    
    # Calcular Kappa entre pares
    kappa_12 = cohen_kappa_score(etiquetas_validador_1, etiquetas_validador_2)
    kappa_13 = cohen_kappa_score(etiquetas_validador_1, etiquetas_validador_3)
    kappa_23 = cohen_kappa_score(etiquetas_validador_2, etiquetas_validador_3)
    
    kappa_promedio = (kappa_12 + kappa_13 + kappa_23) / 3
    
    print(f"Kappa de Cohen promedio: {kappa_promedio:.2f}")
    
    if kappa_promedio < 0.7:
        print("⚠️ ALERTA: Inconsistencia en etiquetado. Revisar criterios.")
    
    return kappa_promedio
```

**Impacto:**
- Identifica ambigüedad en criterios de clasificación
- Requiere documentación clara de políticas de etiquetado
- Necesita sesión de alineación con contadores (2-4 horas)
