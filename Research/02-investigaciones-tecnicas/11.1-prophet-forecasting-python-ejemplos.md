## Marco Teórico

### ¿Qué es Prophet?

**Prophet** es un procedimiento de pronóstico de series de tiempo desarrollado por Facebook (Meta) en 2017. Está diseñado para ser fácil de usar, automatizado y robusto para datos del mundo real.

**Características principales:**
- **Modelo aditivo**: Tendencia + Estacionalidad + Holidays
- **Manejo de outliers**: Robusto a valores atípicos
- **Datos faltantes**: Manejo automático
- **Cambios de tendencia**: Detección automática de changepoints
- **Múltiples estacionalidades**: Yearly, weekly, daily

**Fórmula del modelo:**
```
y(t) = g(t) + s(t) + h(t) + ε(t)

Donde:
- g(t): Tendencia (growth)
- s(t): Estacionalidad (seasonality)
- h(t): Holidays
- ε(t): Error (ruido)
```

## Preprocesamiento de Datos

### Formato Requerido por Prophet

Prophet requiere un DataFrame de pandas con dos columnas:

| Columna | Tipo | Descripción | Ejemplo |
|---------|------|-------------|---------|
| **ds** | datetime | Fecha (datestamp) | 2026-01-15 |
| **y** | float | Valor a predecir | 125000.50 |

### Ejemplo de Preprocesamiento

```python
import pandas as pd
from typing import Optional

class PreprocesadorDatosProphet:
    """
    Preprocesa datos para Prophet con formato mexicano.
    """
    
    def __init__(self, timezone: str = 'America/Mexico_City'):
        """
        Inicializa el preprocesador.
        
        Args:
            timezone: Timezone de México (default: America/Mexico_City)
        """
        self.timezone = timezone
    
    def formatear_datos(self, df: pd.DataFrame, 
                        columna_fecha: str, 
                        columna_valor: str,
                        moneda: str = 'MXN') -> pd.DataFrame:
        """
        Formatea datos para Prophet.
        
        Args:
            df: DataFrame con datos históricos
            columna_fecha: Nombre de columna con fechas
            columna_valor: Nombre de columna con valores
        
        Returns:
            DataFrame formateado para Prophet
        """
        # Crear copia
        datos = df.copy()
        
        # Convertir fecha a datetime
        datos[columna_fecha] = pd.to_datetime(datos[columna_fecha])
        
        # Renombrar columnas para Prophet
        datos = datos.rename(columns={
            columna_fecha: 'ds',
            columna_valor: 'y'
        })
        
        # Seleccionar solo columnas requeridas
        datos = datos[['ds', 'y']].copy()
        
        # Ordenar por fecha
        datos = datos.sort_values('ds')
        
        # Eliminar duplicados
        datos = datos.drop_duplicates(subset=['ds'], keep='last')
        
        # Manejar valores nulos
        if datos['y'].isnull().any():
            print(f"Advertencia: {datos['y'].isnull().sum()} valores nulos encontrados")
            # Interpolar valores nulos
            datos['y'] = datos['y'].interpolate(method='linear')
        
        # Convertir timezone
        datos['ds'] = datos['ds'].dt.tz_localize(self.timezone)
        
        return datos
    
    def detectar_outliers(self, df: pd.DataFrame, 
                          metodo: str = 'iqr',
                          multiplicador: float = 1.5) -> pd.DataFrame:
        """
        Detecta outliers en los datos.
        
        Args:
            df: DataFrame con columnas ds, y
            metodo: 'iqr' o 'zscore'
            multiplicador: Multiplicador para detectar outliers
        
        Returns:
            DataFrame con columna 'outlier' (True/False)
        """
        datos = df.copy()
        
        if metodo == 'iqr':
            Q1 = datos['y'].quantile(0.25)
            Q3 = datos['y'].quantile(0.75)
            IQR = Q3 - Q1
            
            limite_inferior = Q1 - multiplicador * IQR
            limite_superior = Q3 + multiplicador * IQR
            
            datos['outlier'] = (datos['y'] < limite_inferior) | (datos['y'] > limite_superior)
        
        elif metodo == 'zscore':
            from scipy import stats
            datos['zscore'] = stats.zscore(datos['y'])
            datos['outlier'] = abs(datos['zscore']) > 3
            datos = datos.drop('zscore', axis=1)
        
        return datos
    
    def imputar_outliers(self, df: pd.DataFrame, 
                         metodo: str = 'mediana') -> pd.DataFrame:
        """
        Imputa outliers con mediana o media.
        
        Args:
            df: DataFrame con columna 'outlier'
            metodo: 'mediana' o 'media'
        
        Returns:
            DataFrame sin outliers
        """
        datos = df.copy()
        
        if metodo == 'mediana':
            valor_imputacion = datos[~datos['outlier']]['y'].median()
        else:
            valor_imputacion = datos[~datos['outlier']]['y'].mean()
        
        datos.loc[datos['outlier'], 'y'] = valor_imputacion
        datos = datos.drop('outlier', axis=1)
        
        return datos


# Ejemplo de uso con datos mexicanos
```

### Configurador de Holidays para Prophet

```python
import pandas as pd
from typing import List, Dict
import holidays

class ConfiguradorHolidaysMexico:
    """
    Configura holidays de México para Prophet.
    """
    
    def __init__(self):
        """Inicializa el configurador de holidays."""
        self.holidays_mexico = holidays.Mexico()
    
    def obtener_holidays_oficiales(self, years: List[int]) -> pd.DataFrame:
        """
        Obtiene holidays oficiales de México.
        
        Args:
            years: Lista de años (ej. [2025, 2026])
        
        Returns:
            DataFrame con columnas ds, holiday
        """
        fechas_holidays = {}
        
        for year in years:
            year_holidays = holidays.Mexico(years=year)
            for date, name in year_holidays.items():
                fechas_holidays[pd.to_datetime(date)] = name
        
        df_holidays = pd.DataFrame({
            'ds': list(fechas_holidays.keys()),
            'holiday': list(fechas_holidays.values())
        })
        
        return df_holidays.sort_values('ds')
    
    def agregar_holidays_comerciales(self, df_holidays: pd.DataFrame,
                                      years: List[int]) -> pd.DataFrame:
        """
        Agrega holidays comerciales relevantes para retail.
        
        Args:
            df_holidays: DataFrame con holidays oficiales
            years: Lista de años
        
        Returns:
            DataFrame con holidays oficiales y comerciales
        """
        holidays_comerciales = {
            'Día de Reyes': [(1, 6)],
            'Día del Amor y la Amistad': [(2, 14)],
            'Día del Niño': [(4, 30)],
            'Día de las Madres': [(5, 10)],
            'Inicio de la Independencia': [(9, 3)],
            'Halloween': [(10, 31)],
            'Día de Muertos': [(11, 2)],
            'Nochebuena': [(12, 24)],
            'Nochevieja': [(12, 31)]
        }
        
        registros_adicionales = []
        
        for year in years:
            for nombre, fechas in holidays_comerciales.items():
                for mes, dia in fechas:
                    try:
                        fecha = pd.to_datetime(f'{year}-{mes:02d}-{dia:02d}')
                        registros_adicionales.append({
                            'ds': fecha,
                            'holiday': nombre
                        })
                    except:
                        pass  # Fecha inválida
        
        df_comerciales = pd.DataFrame(registros_adicionales)
        df_combinado = pd.concat([df_holidays, df_comerciales], ignore_index=True)
        
        return df_combinado.sort_values('ds').reset_index(drop=True)
    
    def crear_holidays_personalizados(self, eventos: List[Dict]) -> pd.DataFrame:
        """
        Crea holidays personalizados (promociones, eventos especiales).
        
        Args:
            eventos: Lista de diccionarios con 'fecha' y 'nombre'
        
        Returns:
            DataFrame con holidays personalizados
        """
        registros = []
        
        for evento in eventos:
            registros.append({
                'ds': pd.to_datetime(evento['fecha']),
                'holiday': evento['nombre']
            })
        
        return pd.DataFrame(registros)
    
    def obtener_holidays_completos(self, years: List[int] = [2025, 2026],
                                    incluir_comerciales: bool = True,
                                    eventos_personalizados: List[Dict] = None) -> pd.DataFrame:
        """
        Obtiene todos los holidays (oficiales, comerciales, personalizados).
        
        Args:
            years: Lista de años
            incluir_comerciales: Si True, incluye holidays comerciales
            eventos_personalizados: Lista de eventos personalizados
        
        Returns:
            DataFrame completo de holidays
        """
        # Holidays oficiales
        df_holidays = self.obtener_holidays_oficiales(years)
        
        # Agregar comerciales
        if incluir_comerciales:
            df_holidays = self.agregar_holidays_comerciales(df_holidays, years)
        
        # Agregar personalizados
        if eventos_personalizados:
            df_personalizados = self.crear_holidays_personalizados(eventos_personalizados)
            df_holidays = pd.concat([df_holidays, df_personalizados], ignore_index=True)
        
        return df_holidays.sort_values('ds').reset_index(drop=True)


# Ejemplo de uso
```

## Configuración del Modelo Prophet

### Parámetros Principales

```python
from prophet import Prophet

model = Prophet(
    # Tendencia
    growth='linear',              # 'linear' o 'logistic'
    changepoints=None,            # Puntos de cambio automáticos
    n_changepoints=25,            # Número de puntos de cambio
    changepoint_range=0.8,        # Rango de puntos de cambio (80% de datos)
    
    # Estacionalidad
    yearly_seasonality=True,      # Estacionalidad anual
    weekly_seasonality=True,      # Estacionalidad semanal
    daily_seasonality=False,      # Estacionalidad diaria (para datos hourly)
    
    # Holidays
    holidays=None,                # DataFrame con holidays
    holidays_prior_scale=10.0,    # Fuerza de holidays
    
    # Priors
    seasonality_prior_scale=10.0, # Fuerza de estacionalidad
    changepoint_prior_scale=0.05, # Flexibilidad de tendencia
    
    # Incertidumbre
    mcmc_samples=0,               # MCMC para incertidumbre (0 = MAP)
    interval_width=0.80,          # Intervalo de confianza (80%)
    uncertainty_samples=1000,     # Muestras para incertidumbre
    
    # Estacionalidades múltiples
    seasonality_mode='additive'   # 'additive' o 'multiplicative'
)
```

```python
from sklearn.model_selection import train_test_split

def dividir_datos(df: pd.DataFrame, 
                  test_size: float = 0.2,
                  shuffle: bool = False) -> tuple:
    """
    Divide datos en train y test para series de tiempo.
    
    Args:
        df: DataFrame con datos
        test_size: Proporción para test (0.2 = 20%)
        shuffle: Si False, mantiene orden temporal
    
    Returns:
        train_df, test_df
    """
    if shuffle:
        train_df, test_df = train_test_split(
            df, 
            test_size=test_size,
            random_state=42
        )
    else:
        # Para series de tiempo, no hacer shuffle
        split_idx = int(len(df) * (1 - test_size))
        train_df = df.iloc[:split_idx].copy()
        test_df = df.iloc[split_idx:].copy()
    
    return train_df, test_df
```

### Cross-Validation para Series de Tiempo

```python
from prophet.diagnostics import cross_validation, performance_metrics

class EntrenadorProphet:
    """
    Entrena modelo Prophet con cross-validation.
    """
    
    def __init__(self, **kwargs_prophet):
        """
        Inicializa el entrenador.
        
        Args:
            **kwargs_prophet: Parámetros para Prophet
        """
        self.model = Prophet(**kwargs_prophet)
        self.df_cv = None
        self.df_metrics = None
    
    def entrenar(self, df: pd.DataFrame,
                 holidays: pd.DataFrame = None) -> 'EntrenadorProphet':
        """
        Entrena el modelo Prophet.
        
        Args:
            df: DataFrame con columnas ds, y
            holidays: DataFrame con holidays
        
        Returns:
            self
        """
        if holidays is not None:
            self.model.add_country_holidays('MX')
            self.model.holidays = holidays
        
        self.model.fit(df)
        return self
    
    def cross_validation(self, df: pd.DataFrame,
                         initial: str = '365 days',
                         period: str = '30 days',
                         horizon: str = '90 days') -> pd.DataFrame:
        """
        Realiza cross-validation.
        
        Args:
            df: DataFrame con datos
            initial: Período inicial de entrenamiento
            period: Período entre cortes
            horizon: Horizonte de predicción
        
        Returns:
            DataFrame con resultados de CV
        """
        self.df_cv = cross_validation(
            self.model,
            initial=initial,
            period=period,
            horizon=horizon
        )
        
        return self.df_cv
    
    def obtener_metricas(self) -> pd.DataFrame:
        """
        Calcula métricas de rendimiento.
        
        Returns:
            DataFrame con métricas (MAPE, RMSE, MAE)
        """
        if self.df_cv is None:
            raise ValueError("Primero ejecuta cross_validation")
        
        self.df_metrics = performance_metrics(self.df_cv)
        return self.df_metrics
    
    def obtener_forecast(self, periods: int = 90,
                         freq: str = 'D') -> pd.DataFrame:
        """
        Genera forecast futuro.
        
        Args:
            periods: Número de períodos a predecir
            freq: Frecuencia ('D', 'W', 'M')
        
        Returns:
            DataFrame con forecast
        """
        future = self.model.make_future_dataframe(
            periods=periods,
            freq=freq
        )
        
        forecast = self.model.predict(future)
        return forecast
```

## Métricas de Evaluación

### Fórmulas

| Métrica | Fórmula | Interpretación | Óptimo |
|---------|---------|----------------|--------|
| **MAPE** | `mean(|(y - ŷ) / y|) × 100` | Error porcentual absoluto | <10% |
| **RMSE** | `sqrt(mean((y - ŷ)²))` | Error cuadrático medio | Minimizar |
| **MAE** | `mean(|y - ŷ|)` | Error absoluto medio | Minimizar |

### Implementación en Python

```python
from typing import Tuple
import numpy as np

class EvaluadorModelo:
    """
    Evalúa modelo con métricas MAPE, RMSE, MAE.
    """
    
    @staticmethod
    def calcular_mape(y_true: np.ndarray, 
                      y_pred: np.ndarray) -> float:
        """
        Calcula MAPE (Mean Absolute Percentage Error).
        
        Args:
            y_true: Valores reales
            y_pred: Valores predichos
        
        Returns:
            MAPE en porcentaje
        """
        y_true = np.array(y_true)
        y_pred = np.array(y_pred)
        
        # Evitar división por cero
        mask = y_true != 0
        
        mape = np.mean(np.abs((y_true[mask] - y_pred[mask]) / y_true[mask])) * 100
        return mape
    
    @staticmethod
    def calcular_rmse(y_true: np.ndarray, 
                      y_pred: np.ndarray) -> float:
        """
        Calcula RMSE (Root Mean Square Error).
        
        Args:
            y_true: Valores reales
            y_pred: Valores predichos
        
        Returns:
            RMSE
        """
        y_true = np.array(y_true)
        y_pred = np.array(y_pred)
        
        rmse = np.sqrt(np.mean((y_true - y_pred) ** 2))
        return rmse
    
    @staticmethod
    def calcular_mae(y_true: np.ndarray, 
                     y_pred: np.ndarray) -> float:
        """
        Calcula MAE (Mean Absolute Error).
        
        Args:
            y_true: Valores reales
            y_pred: Valores predichos
        
        Returns:
            MAE
        """
        y_true = np.array(y_true)
        y_pred = np.array(y_pred)
        
        mae = np.mean(np.abs(y_true - y_pred))
        return mae
    
    @staticmethod
    def evaluar_completo(y_true: np.ndarray, 
                         y_pred: np.ndarray) -> dict:
        """
        Calcula todas las métricas.
        
        Args:
            y_true: Valores reales
            y_pred: Valores predichos
        
        Returns:
            Diccionario con todas las métricas
        """
        return {
            'mape': EvaluadorModelo.calcular_mape(y_true, y_pred),
            'rmse': EvaluadorModelo.calcular_rmse(y_true, y_pred),
            'mae': EvaluadorModelo.calcular_mae(y_true, y_pred),
            'mape_interpretacion': EvaluadorModelo.interpretar_mape(
                EvaluadorModelo.calcular_mape(y_true, y_pred)
            )
        }
    
    @staticmethod
    def interpretar_mape(mape: float) -> str:
        """
        Interpreta el valor de MAPE.
        
        Args:
            mape: Valor de MAPE en porcentaje
        
        Returns:
            Interpretación del MAPE
        """
        if mape < 5:
            return "Excelente (error < 5%)"
        elif mape < 10:
            return "Muy bueno (error 5-10%)"
        elif mape < 15:
            return "Aceptable (error 10-15%)"
        elif mape < 20:
            return "Regular (error 15-20%)"
        else:
            return "Pobre (error > 20%)"


# Ejemplo de uso
if __name__ == "__main__":
    # Datos de prueba
    y_real = test_df['y'].values
    y_predicho = forecast.loc[forecast['ds'].isin(test_df['ds']), 'yhat'].values
    
    # Evaluar
    evaluador = EvaluadorModelo()
    metricas = evaluador.evaluar_completo(y_real, y_predicho)
    
    print(f"MAPE: {metricas['mape']:.2f}% - {metricas['mape_interpretacion']}")
    print(f"RMSE: ${metricas['rmse']:,.2f} MXN")
    print(f"MAE: ${metricas['mae']:,.2f} MXN")
```

## Generación de Forecast y Visualización

```python
import matplotlib.pyplot as plt
from prophet.plot import plot_plotly, plot_components_plotly

class GeneradorForecast:
    """
    Genera forecast y visualizaciones.
    """
    
    def __init__(self, model: Prophet, forecast: pd.DataFrame):
        """
        Inicializa el generador.
        
        Args:
            model: Modelo Prophet entrenado
            forecast: DataFrame con forecast
        """
        self.model = model
        self.forecast = forecast
    
    def plot_forecast(self, show_plot: bool = True,
                      save_path: str = None) -> plt.Figure:
        """
        Grafica forecast histórico y futuro.
        
        Args:
            show_plot: Si True, muestra el gráfico
            save_path: Ruta para guardar el gráfico
        
        Returns:
            Figura de matplotlib
        """
        fig, ax = plt.subplots(figsize=(14, 7))
        
        # Plot histórico
        ax.plot(
            self.model.history['ds'],
            self.model.history['y'],
            'k.',
            label='Real'
        )
        
        # Plot forecast
        ax.plot(
            self.forecast['ds'],
            self.forecast['yhat'],
            'b-',
            label='Forecast',
            linewidth=2
        )
        
        # Intervalo de confianza
        ax.fill_between(
            self.forecast['ds'],
            self.forecast['yhat_lower'],
            self.forecast['yhat_upper'],
            color='blue',
            alpha=0.2,
            label='Intervalo 80%'
        )
        
        ax.set_xlabel('Fecha', fontsize=12)
        ax.set_ylabel('Ventas (MXN)', fontsize=12)
        ax.set_title('Forecast de Ventas con Prophet', fontsize=14)
        ax.legend(loc='upper left')
        ax.grid(True, alpha=0.3)
        
        plt.xticks(rotation=45)
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
        
        if show_plot:
            plt.show()
        
        return fig
    
    def plot_componentes(self, show_plot: bool = True) -> plt.Figure:
        """
        Grafica componentes del modelo.
        
        Args:
            show_plot: Si True, muestra el gráfico
        
        Returns:
            Figura de matplotlib
        """
        fig = self.model.plot_components(self.forecast)
        
        if show_plot:
            plt.show()
        
        return fig
    
    def obtener_resumen(self) -> pd.DataFrame:
        """
        Obtiene resumen del forecast.
        
        Returns:
            DataFrame con resumen estadístico
        """
        resumen = self.forecast[['ds', 'yhat', 'yhat_lower', 'yhat_upper']].copy()
        resumen['tendencia'] = self.forecast['trend']
        resumen['estacionalidad'] = self.forecast['weekly'] + self.forecast['yearly']
        
        return resumen


# Ejemplo de uso
if __name__ == "__main__":
    generador = GeneradorForecast(entrenador.model, forecast)
    
    # Plot forecast
    generador.plot_forecast(save_path='forecast_ventas.png')
    
    # Plot componentes
    generador.plot_componentes()
    
    # Resumen
    resumen = generador.obtener_resumen()
    print(resumen.tail(10))
```

```python
# Datos con estacionalidad de diciembre
ventas_retail = {
    'fecha': pd.date_range(start='2024-01-01', end='2026-02-28', freq='D'),
    'ventas': []
}

for fecha in ventas_retail['fecha']:
    base = 100000
    tendencia = (fecha - pd.to_datetime('2024-01-01')).days * 50
    
    # Estacionalidad semanal
    if fecha.dayofweek >= 5:  # Fin de semana
        semanal = 30000
    else:
        semanal = 0
    
    # Estacionalidad anual (diciembre)
    if fecha.month == 12:
        anual = 150000
    else:
        anual = 0
    
    # Holidays
    if fecha.month == 12 and fecha.day in [24, 25, 31]:
        holiday = 200000
    else:
        holiday = 0
    
    ventas = base + tendencia + semanal + anual + holiday
    ventas_retail['ventas'].append(ventas + np.random.normal(0, 10000))

df_retail = pd.DataFrame(ventas_retail)

# Entrenar modelo
preprocesador = PreprocesadorDatosProphet()
datos_prophet = preprocesador.formatear_datos(df_retail, 'fecha', 'ventas')

configurador = ConfiguradorHolidaysMexico()
holidays = configurador.obtener_holidays_completos(years=[2024, 2025, 2026])

entrenador = EntrenadorProphet(
    growth='linear',
    yearly_seasonality=True,
    weekly_seasonality=True,
    holidays=holidays,
    holidays_prior_scale=20.0  # Holidays más fuertes para retail
)

entrenador.entrenar(datos_prophet)

# Forecast para diciembre 2026
forecast = entrenador.obtener_forecast(periods=365, freq='D')

# Filtrar diciembre
diciembre_2026 = forecast[forecast['ds'].dt.month == 12]
```

### Caso 2: Forecasting de Costos (Inflación)

```python
# Proyectar costos con inflación
inflacion_anual = 0.045  # 4.5% inflación esperada México 2026

costos_mensuales = {
    'fecha': pd.date_range(start='2024-01-01', end='2026-02-28', freq='M'),
    'costos': []
}

base_costos = 500000
for i, fecha in enumerate(costos_mensuales['fecha']):
    # Inflación acumulada
    inflacion_acumulada = (1 + inflacion_anual) ** (i / 12)
    costos = base_costos * inflacion_acumulada
    costos_mensuales['costos'].append(costos + np.random.normal(0, 20000))

df_costos = pd.DataFrame(costos_mensuales)

# Modelo con crecimiento logístico (saturación)
datos_prophet = preprocesador.formatear_datos(df_costos, 'fecha', 'costos')

model_costos = Prophet(
    growth='logistic',
    yearly_seasonality=True,
    changepoint_prior_scale=0.01  # Menos flexible
)

# Agregar capacidad máxima
datos_prophet['capacidad'] = 1000000  # Límite de costos

model_costos.fit(datos_prophet)

future = model_costos.make_future_dataframe(periods=12, freq='M')
future['capacidad'] = 1000000

forecast_costos = model_costos.predict(future)
print(f"Costos proyectados 2027: ${forecast_costos['yhat'].iloc[-12:].sum():,.2f} MXN")
```
