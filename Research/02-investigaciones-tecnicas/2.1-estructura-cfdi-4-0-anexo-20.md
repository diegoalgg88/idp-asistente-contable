### 3.2 Algoritmos Específicos

#### Algoritmo 1: Validación de Estructura XML contra XSD

```python
from lxml import etree
from typing import Dict, List, Tuple
import os

class CFDI40Validator:
    """
    Validador de CFDI 4.0 contra esquemas XSD del SAT.
    Implementa validación en 3 niveles: cfdi40.xsd, tipos.xsd, catálogos.xsd
    """
    
    def __init__(self, xsd_path: str = "schemas/cfdi40/"):
        """
        Inicializa el validador con rutas a esquemas XSD.
        
        Args:
            xsd_path: Ruta al directorio con esquemas XSD del SAT
        """
        self.xsd_path = xsd_path
        self.cfdi_schema = self._load_schema("cfdi40.xsd")
        self.tipos_schema = self._load_schema("tipos.xsd")
        self.catalogos_schema = self._load_schema("catálogos.xsd")
    
    def _load_schema(self, filename: str) -> etree.XMLSchema:
        """
        Carga un esquema XSD desde archivo.
        
        Args:
            filename: Nombre del archivo XSD
            
        Returns:
            XMLSchema cargado
        """
        schema_path = os.path.join(self.xsd_path, filename)
        with open(schema_path, 'r', encoding='utf-8') as f:
            schema_root = etree.XML(f.read())
        return etree.XMLSchema(schema_root)
    
    def validate_cfdi(self, xml_content: str) -> Dict:
        """
        Valida un CFDI 4.0 completo contra esquemas XSD.
        
        Args:
            xml_content: Contenido XML del CFDI como string
            
        Returns:
            Dict con resultados de validación:
            - is_valid: bool
            - errors: List[str]
            - warnings: List[str]
        """
        errors = []
        warnings = []
        
        try:
            # Parsear XML
            cfdi_doc = etree.XML(xml_content.encode('utf-8'))
            
            # Validar contra cfdi40.xsd
            if not self.cfdi_schema.validate(cfdi_doc):
                for error in self.cfdi_schema.error_log:
                    errors.append(f"Línea {error.line}: {error.message}")
            
            # Validar campos obligatorios
            obligatorios_errors = self._validate_campos_obligatorios(cfdi_doc)
            errors.extend(obligatorios_errors)
            
            # Validar catálogos
            catalogos_errors = self._validate_catalogos(cfdi_doc)
            errors.extend(catalogos_errors)
            
            return {
                'is_valid': len(errors) == 0,
                'errors': errors,
                'warnings': warnings,
                'schema_version': '4.0',
                'validated_at': datetime.now().isoformat()
            }
            
        except etree.XMLSyntaxError as e:
            return {
                'is_valid': False,
                'errors': [f"Error de sintaxis XML: {str(e)}"],
                'warnings': [],
                'schema_version': '4.0',
                'validated_at': datetime.now().isoformat()
            }
    
    def _validate_campos_obligatorios(self, cfdi_doc: etree._Element) -> List[str]:
        """
        Valida que todos los campos obligatorios del CFDI 4.0 estén presentes.
        
        Args:
            cfdi_doc: Documento XML parseado
            
        Returns:
            Lista de errores encontrados
        """
        errors = []
        
        # Comprobante (atributos obligatorios)
        comprobante = cfdi_doc.find('.//cfdi:Comprobante', namespaces={'cfdi': 'http://www.sat.gob.mx/cfd/4'})
        if comprobante is not None:
            campos_obligatorios = [
                'Version', 'Serie', 'Folio', 'Fecha', 'SelloCFD', 'NoCertificado',
                'Certificado', 'SubTotal', 'Total', 'TipoDeComprobante', 'Exportacion',
                'Moneda', 'TipoCambio', 'MetodoPago', 'FormaPago', 'LugarExpedicion'
            ]
            
            for campo in campos_obligatorios:
                if campo not in comprobante.attrib:
                    errors.append(f"Campo obligatorio faltante en Comprobante: {campo}")
        
        # Emisor
        emisor = cfdi_doc.find('.//cfdi:Emisor', namespaces={'cfdi': 'http://www.sat.gob.mx/cfd/4'})
        if emisor is not None:
            if 'Rfc' not in emisor.attrib:
                errors.append("Campo obligatorio faltante: Emisor.Rfc")
            if 'Nombre' not in emisor.attrib:
                errors.append("Campo obligatorio faltante: Emisor.Nombre")
            if 'RegimenFiscal' not in emisor.attrib:
                errors.append("Campo obligatorio faltante: Emisor.RegimenFiscal")
        
        # Receptor
        receptor = cfdi_doc.find('.//cfdi:Receptor', namespaces={'cfdi': 'http://www.sat.gob.mx/cfd/4'})
        if receptor is not None:
            campos_receptor = ['Rfc', 'Nombre', 'DomicilioFiscalReceptor', 'RegimenFiscal', 'UsoCFDI']
            for campo in campos_receptor:
                if campo not in receptor.attrib:
                    errors.append(f"Campo obligatorio faltante: Receptor.{campo}")
        
        # Conceptos (al menos 1)
        conceptos = cfdi_doc.findall('.//cfdi:Concepto', namespaces={'cfdi': 'http://www.sat.gob.mx/cfd/4'})
        if len(conceptos) == 0:
            errors.append("Se requiere al menos 1 concepto en el CFDI")
        
        # Validar cada concepto
        for i, concepto in enumerate(conceptos):
            campos_concepto = [
                'ClaveProdServ', 'NoIdentificacion', 'Cantidad', 'ClaveUnidad',
                'Descripcion', 'ValorUnitario', 'Importe', 'ObjetoImp'
            ]
            for campo in campos_concepto:
                if campo not in concepto.attrib:
                    errors.append(f"Campo obligatorio faltante en Concepto {i+1}: {campo}")
        
        # Impuestos (si ObjetoImp = "02")
        for concepto in conceptos:
            if concepto.attrib.get('ObjetoImp') == '02':  # Sí objeto de impuesto
                impuestos = concepto.find('.//cfdi:Impuestos', namespaces={'cfdi': 'http://www.sat.gob.mx/cfd/4'})
                if impuestos is None:
                    errors.append(f"Concepto {concepto.attrib.get('ClaveProdServ')}: ObjetoImp=02 requiere nodo Impuestos")
        
        return errors
    
    def _validate_catalogos(self, cfdi_doc: etree._Element) -> List[str]:
        """
        Valida que las claves de catálogos del SAT sean válidas.
        
        Args:
            cfdi_doc: Documento XML parseado
            
        Returns:
            Lista de errores de catálogo
        """
        errors = []
        
        # Cargar catálogos SAT (desde archivos JSON/Excel)
        catalogos = self._load_catalogos_sat()
        
        # Validar ClaveProdServ
        conceptos = cfdi_doc.findall('.//cfdi:Concepto', namespaces={'cfdi': 'http://www.sat.gob.mx/cfd/4'})
        for concepto in conceptos:
            clave_prod_serv = concepto.attrib.get('ClaveProdServ')
            if clave_prod_serv and clave_prod_serv not in catalogos['clave_prod_serv']:
                errors.append(f"ClaveProdServ inválida: {clave_prod_serv}")
        
        # Validar ClaveUnidad
        for concepto in conceptos:
            clave_unidad = concepto.attrib.get('ClaveUnidad')
            if clave_unidad and clave_unidad not in catalogos['clave_unidad']:
                errors.append(f"ClaveUnidad inválida: {clave_unidad}")
        
        # Validar UsoCFDI
        receptor = cfdi_doc.find('.//cfdi:Receptor', namespaces={'cfdi': 'http://www.sat.gob.mx/cfd/4'})
        if receptor is not None:
            uso_cfdi = receptor.attrib.get('UsoCFDI')
            if uso_cfdi and uso_cfdi not in catalogos['uso_cfdi']:
                errors.append(f"UsoCFDI inválido: {uso_cfdi}")
        
        # Validar RegimenFiscal (emisor y receptor)
        for nodo in ['Emisor', 'Receptor']:
            elemento = cfdi_doc.find(f'.//cfdi:{nodo}', namespaces={'cfdi': 'http://www.sat.gob.mx/cfd/4'})
            if elemento is not None:
                regimen = elemento.attrib.get('RegimenFiscal')
                if regimen and regimen not in catalogos['regimen_fiscal']:
                    errors.append(f"{nodo}.RegimenFiscal inválido: {regimen}")
        
        return errors
    
    def _load_catalogos_sat(self) -> Dict[str, set]:
        """
        Carga catálogos oficiales del SAT.
        
        Returns:
            Dict con conjuntos de claves válidas por catálogo
        """
        # Implementación: cargar desde archivos JSON/Excel del SAT
        # URL oficial: https://www.sat.gob.mx/aplicacion/29414/consulta-tu-proveedor-de-certificacion
        return {
            'clave_prod_serv': set(),  # 50,000+ claves
            'clave_unidad': set(),     # 100+ claves
            'uso_cfdi': set(),         # 50+ claves
            'regimen_fiscal': set()    # 30+ claves
        }
```

#### Algoritmo 2: Generación de Cadena Original

```python
from lxml import etree
import base64
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import padding
from cryptography.hazmat.backends import default_backend

class CadenaOriginalGenerator:
    """
    Genera la cadena original del CFDI 4.0 para sellado digital.
    Usa el XSLT oficial del SAT (cadenaoriginal_4_0.xslt)
    """
    
    def __init__(self, xslt_path: str = "xslt/cadenaoriginal_4_0.xslt"):
        """
        Inicializa el generador con XSLT oficial del SAT.
        
        Args:
            xslt_path: Ruta al archivo XSLT de cadena original
        """
        with open(xslt_path, 'r', encoding='utf-8') as f:
            xslt_content = f.read()
        self.xslt_transform = etree.XSLT(etree.XML(xslt_content.encode('utf-8')))
    
    def generate(self, cfdi_xml: str) -> str:
        """
        Genera la cadena original a partir del XML del CFDI.
        
        Args:
            cfdi_xml: Contenido XML del CFDI
            
        Returns:
            Cadena original lista para sellado
        """
        # Parsear XML
        cfdi_doc = etree.XML(cfdi_xml.encode('utf-8'))
        
        # Aplicar transformación XSLT
        cadena_original = self.xslt_transform(cfdi_doc)
        
        return str(cadena_original)
    
    def sign(self, cadena_original: str, private_key_path: str, password: str) -> str:
        """
        Firma la cadena original usando la llave privada del CSD.
        Algoritmo: RSA-SHA256
        
        Args:
            cadena_original: Cadena original a firmar
            private_key_path: Ruta a la llave privada (.key)
            password: Contraseña de la llave privada
            
        Returns:
            Sello digital en Base64
        """
        # Cargar llave privada
        with open(private_key_path, 'rb') as f:
            private_key = serialization.load_pem_private_key(
                f.read(),
                password=password.encode('utf-8'),
                backend=default_backend()
            )
        
        # Calcular hash SHA-256
        digest = hashes.Hash(hashes.SHA256(), backend=default_backend())
        digest.update(cadena_original.encode('utf-8'))
        digest_value = digest.finalize()
        
        # Firmar con RSA
        signature = private_key.sign(
            digest_value,
            padding.PKCS1v15(),
            hashes.SHA256()
        )
        
        # Convertir a Base64
        sello_base64 = base64.b64encode(signature).decode('utf-8')
        
        return sello_base64
```

    cadena_original_timbre = _generate_timbre_cadena_original(timbre)

## 4. Limitantes y Restricciones

### 4.1 Limitación 1: Sin API Oficial del SAT para Timbrado

**Problema:**
El SAT no proporciona una API directa para timbrado de CFDI. Es obligatorio usar un PAC (Proveedor Autorizado de Certificación) autorizado.

**Solución:**
```python
# Abstracción de múltiples PACs para fallback automático
class PACManager:
    def __init__(self):
        self.pacs = {
            'finkok': FinkokPAC(api_key, secret),
            'sw_sapien': SWPAC(token),
            'ecodex': EcodexPAC(oauth_client, oauth_secret),
            'solucion_factible': SFPAC(api_key)
        }
        self.primary_pac = 'finkok'
    
    def stamp(self, cfdi_xml: str) -> Dict:
        """
        Intenta timbrar con PAC primario, fallback a secundarios.
        """
        for pac_name in [self.primary_pac] + list(self.pacs.keys()):
            try:
                pac = self.pacs[pac_name]
                result = pac.stamp(cfdi_xml)
                if result['success']:
                    return result
            except Exception as e:
                logger.warning(f"PAC {pac_name} falló: {str(e)}")
                continue
        
        raise Exception("Todos los PACs fallaron")
```

**Impacto:**
- Costo adicional: $0.70-$3.50 MXN por timbre
- Dependencia de terceros (PACs)
- SLA variable entre PACs (99.5%-99.9%)

### 4.2 Limitación 2: Cambios en Catálogos SAT sin Aviso

**Problema:**
El SAT actualiza catálogos (ClaveProdServ, UsoCFDI, etc.) sin previo aviso. Una clave inválida causa rechazo inmediato.

**Solución:**
```python
# Actualización automática de catálogos desde SAT
class SATCatalogosUpdater:
    SAT_URL_BASE = "https://www.sat.gob.mx/aplicacion/29414/consulta-tu-proveedor-de-certificacion"
    
    def __init__(self, db_session):
        self.db = db_session
    
    def check_and_update(self) -> Dict:
        """
        Verifica si hay nuevos catálogos y actualiza base de datos local.
        Se ejecuta diariamente a las 6:00 AM.
        """
        catalogos = {
            'clave_prod_serv': f"{self.SAT_URL_BASE}/catalogos/ClaveProdServ.csv",
            'clave_unidad': f"{self.SAT_URL_BASE}/catalogos/ClaveUnidad.csv",
            'uso_cfdi': f"{self.SAT_URL_BASE}/catalogos/UsoCFDI.csv",
            'regimen_fiscal': f"{self.SAT_URL_BASE}/catalogos/RegimenFiscal.csv"
        }
        
        for nombre, url in catalogos.items():
            response = requests.get(url, timeout=30)
            if response.status_code == 200:
                # Parsear CSV y actualizar base de datos local
                self._update_catalogo_db(nombre, response.text)
        
        return {'updated': True, 'timestamp': datetime.now().isoformat()}
```

**Impacto:**
- Requiere monitoreo diario del portal SAT
- Validaciones pueden fallar si catálogos están desactualizados
- Necesidad de caché local para performance

## 8. Conclusiones y Recomendaciones

### 8.1 Hallazgos Clave

1. **Anexo 20 es un estándar cerrado:** No hay flexibilidad en estructura XML. Validación XSD es obligatoria.
2. **50+ campos obligatorios:** Cualquier campo faltante causa rechazo inmediato del PAC.
3. **Catálogos dinámicos:** El SAT actualiza catálogos sin previo aviso. Actualización automática es crítica.
4. **Timbrado requiere PAC:** No hay API directa del SAT. Múltiples PACs proporcionan resiliencia.
5. **Cadena original + sello:** Algoritmo RSA-SHA256 es crítico. Librerías criptográficas deben ser precisas.
