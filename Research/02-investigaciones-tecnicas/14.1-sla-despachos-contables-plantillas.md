## ¿Qué es un SLA Contable?

### Definición

**SLA** (Service Level Agreement / Acuerdo de Nivel de Servicio): Contrato entre un despacho contable y su cliente que define los niveles mínimos de servicio esperados, métricas de medición, responsabilidades de ambas partes, y penalizaciones por incumplimiento.

### Componentes Principales

| Componente | Descripción | Ejemplo |
|------------|-------------|---------|
| **Servicios** | Catálogo de servicios incluidos | Contabilidad, nómina, impuestos |
| **Tiempos de respuesta** | Plazos máximos de atención | 24 hrs para consultas |
| **Disponibilidad** | Porcentaje de tiempo operativo | 99.5% en temporada fiscal |
| **Métricas de calidad** | KPIs de cumplimiento | 98% de declaraciones presentadas a tiempo |
| **Penalizaciones** | Consecuencias por incumplimiento | 5% de descuento por mes |
| **Exclusiones** | Servicios no incluidos | Auditoría externa, litigios |

## Métricas de Cumplimiento (KPIs)

```python
"""
Módulo: calculadora_sla_contable.py
Propósito: Calcular métricas de cumplimiento de SLA para despachos contables
Versión: 1.0
Fecha: 2026-03-10
"""

from dataclasses import dataclass
from typing import Dict, List, Optional
from decimal import Decimal
from datetime import datetime, timedelta


@dataclass
class MetricaSLA:
    """Representa una métrica de SLA."""
    nombre: str
    descripcion: str
    objetivo: float  # Porcentaje objetivo (ej. 98.5)
    actual: float  # Porcentaje actual logrado
    peso: float  # Peso en cálculo de penalización (0-1)
    unidad: str  # '%', 'horas', 'días'


class CalculadoraMetricasSLA:
    """
    Calcula métricas de cumplimiento de SLA para despachos contables.
    """
    
    def __init__(self):
        """Inicializa la calculadora de métricas."""
        self.metricas: List[MetricaSLA] = []
    
    def agregar_metrica(self, metrica: MetricaSLA):
        """
        Agrega una métrica al cálculo.
        
        Args:
            metrica: Objeto MetricaSLA
        """
        self.metricas.append(metrica)
    
    def calcular_cumplimiento_individual(self, metrica: MetricaSLA) -> Dict:
        """
        Calcula cumplimiento individual de una métrica.
        
        Args:
            metrica: Métrica a evaluar
        
        Returns:
            Diccionario con resultado del cálculo
        """
        cumplimiento = (metrica.actual / metrica.objetivo) * 100 if metrica.objetivo > 0 else 0
        
        if cumplimiento >= 100:
            estatus = '✅ Cumplido'
            penalizacion = 0
        elif cumplimiento >= 90:
            estatus = '⚠️ Parcial'
            penalizacion = (100 - cumplimiento) * 0.1  # 10% de penalización por cada 1%
        else:
            estatus = '❌ Incumplido'
            penalizacion = (100 - cumplimiento) * 0.2  # 20% de penalización por cada 1%
        
        return {
            'metrica': metrica.nombre,
            'objetivo': f'{metrica.objetivo}{metrica.unidad}',
            'actual': f'{metrica.actual}{metrica.unidad}',
            'cumplimiento': f'{cumplimiento:.1f}%',
            'estatus': estatus,
            'penalizacion_aplicable': f'{penalizacion:.2f}%'
        }
    
    def calcular_cumplimiento_global(self) -> Dict:
        """
        Calcula cumplimiento global ponderado de todas las métricas.
        
        Returns:
            Diccionario con cumplimiento global y penalización total
        """
        if not self.metricas:
            return {'error': 'No hay métricas registradas'}
        
        cumplimiento_ponderado = 0
        peso_total = sum(m.peso for m in self.metricas)
        penalizacion_total = 0
        
        detalles = []
        
        for metrica in self.metricas:
            resultado = self.calcular_cumplimiento_individual(metrica)
            detalles.append(resultado)
            
            # Contribución al cumplimiento global (ponderado)
            cumplimiento_pct = float(resultado['cumplimiento'].replace('%', ''))
            contribucion = cumplimiento_pct * (metrica.peso / peso_total)
            cumplimiento_ponderado += contribucion
            
            # Contribución a penalización (ponderado)
            penalizacion_pct = float(resultado['penalizacion_aplicable'].replace('%', ''))
            penalizacion_total += penalizacion_pct * (metrica.peso / peso_total)
        
        # Determinar estatus global
        if cumplimiento_ponderado >= 98:
            estatus_global = '✅ Excelente'
        elif cumplimiento_ponderado >= 95:
            estatus_global = '✅ Bueno'
        elif cumplimiento_ponderado >= 90:
            estatus_global = '⚠️ Aceptable'
        else:
            estatus_global = '❌ Deficiente'
        
        return {
            'cumplimiento_global': f'{cumplimiento_ponderado:.2f}%',
            'estatus_global': estatus_global,
            'penalizacion_total': f'{penalizacion_total:.2f}%',
            'total_metricas': len(self.metricas),
            'detalle_metricas': detalles
        }
    
    def generar_reporte_mensual(self, cliente: str, mes: int, anio: int) -> str:
        """
        Genera reporte mensual de cumplimiento de SLA.
        
        Args:
            cliente: Nombre del cliente
            mes: Mes del reporte
            anio: Año del reporte
        
        Returns:
            Reporte en formato texto
        """
        resultado = self.calcular_cumplimiento_global()
        
        reporte = f"""
================================================================================
                    REPORTE DE CUMPLIMIENTO DE SLA
================================================================================
Cliente: {cliente}
Período: {mes:02d}/{anio}
================================================================================

RESUMEN EJECUTIVO
-----------------
Cumplimiento Global: {resultado['cumplimiento_global']}
Estatus: {resultado['estatus_global']}
Penalización Aplicable: {resultado['penalizacion_total']}

MÉTRICAS INDIVIDUALES
---------------------
"""
        
        for metrica in resultado['detalle_metricas']:
            reporte += f"""
{metrica['metrica']}
  Objetivo: {metrica['objetivo']}
  Actual: {metrica['actual']}
  Cumplimiento: {metrica['cumplimiento']}
  Estatus: {metrica['estatus']}
  Penalización: {metrica['penalizacion_aplicable']}
"""
        
        reporte += """
================================================================================
"""
        
        return reporte


# Ejemplo de uso
if __name__ == "__main__":

    # Agregar métricas típicas de despacho contable

        ),
        MetricaSLA(
            nombre='Exactitud en nómina',

            objetivo=99.5,

        ),
```

## Penalizaciones por Incumplimiento

### Estructura Típica de Penalizaciones

| Nivel de Incumplimiento | Penalización | Aplicación |
|------------------------|--------------|------------|
| **98-100% cumplimiento** | 0% | Sin penalización |
| **95-97.9% cumplimiento** | 2-5% | Descuento en factura mensual |
| **90-94.9% cumplimiento** | 5-10% | Descuento + plan de mejora |
| **<90% cumplimiento** | 10-20% | Descuento + revisión de contrato |

### Ejemplo de Cálculo de Penalización

```python
def calcular_penalizacion(honorarios_mensuales: float, 
                          cumplimiento_global: float) -> Dict:
    """
    Calcula penalización por incumplimiento de SLA.
    
    Args:
        honorarios_mensuales: Honorarios mensuales del servicio
        cumplimiento_global: Porcentaje de cumplimiento global
    
    Returns:
        Diccionario con cálculo de penalización
    """
    # Determinar porcentaje de penalización
    if cumplimiento_global >= 98:
        porcentaje_penalizacion = 0
        concepto = 'Sin penalización (cumplimiento excelente)'
    elif cumplimiento_global >= 95:
        porcentaje_penalizacion = 2.5
        concepto = 'Penalización leve (cumplimiento bueno)'
    elif cumplimiento_global >= 90:
        porcentaje_penalizacion = 7.5
        concepto = 'Penalización moderada (cumplimiento aceptable)'
    else:
        porcentaje_penalizacion = 15.0
        concepto = 'Penalización grave (cumplimiento deficiente)'
    
    # Calcular monto
    monto_penalizacion = honorarios_mensuales * (porcentaje_penalizacion / 100)
    honorarios_netos = honorarios_mensuales - monto_penalizacion
    
    return {
        'honorarios_brutos': f'${honorarios_mensuales:,.2f} MXN',
        'cumplimiento_global': f'{cumplimiento_global:.2f}%',
        'porcentaje_penalizacion': f'{porcentaje_penalizacion:.1f}%',
        'monto_penalizacion': f'${monto_penalizacion:,.2f} MXN',
        'honorarios_netos': f'${honorarios_netos:,.2f} MXN',
        'concepto': concepto
    }


# Ejemplo
penalizacion = calcular_penalizacion(
    honorarios_mensuales=50000,
    cumplimiento_global=94.5
)

print("=== Cálculo de Penalización ===")
print(f"Honorarios brutos: {penalizacion['honorarios_brutos']}")
print(f"Cumplimiento global: {penalizacion['cumplimiento_global']}")
print(f"Penalización: {penalizacion['porcentaje_penalizacion']} ({penalizacion['concepto']})")
print(f"Monto penalización: {penalizacion['monto_penalizacion']}")
print(f"Honorarios netos: {penalizacion['honorarios_netos']}")
```
